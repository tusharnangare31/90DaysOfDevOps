git --version - check the version of git installed on the system
git config --global user.name "Your Name" - set the global username for git
git config --global user.email "your.email@example.com" - set the global email for git
git init - intialize a new git repository
git add <file-name> - add a file to the staging area
git commit -m "commit message" - commit the changes in the staging area with a message
git status - check the status of the repository
git restore --staged <file-name> - unstage a file from the staging area
git log - view the commit history
git log --oneline - view the commit history in a short format
git log --graph --oneline - view the commit history in a graphical format
git branch - view the branches in the repository
git checkout -b devops - create a new branch named devops and switch to it
git checkout master - switch back to the master branch
git merge devops - merge the devops branch into the master branch
git branch -M main - rename the current branch to main
git remote add origin <repository-url> - add a remote repository
git push -u origin main - push the changes to the remote repository and set the upstream branch
git pull origin main - pull the latest changes from the remote repository
git clone <repository-url> - clone a remote repository to the local machine
git diff - view the changes made to the files
